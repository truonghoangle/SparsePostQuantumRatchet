use spqr::encoding::polynomial::{PolyDecoder, PolyEncoder};
use spqr::encoding::{Chunk, Decoder, Encoder};
use spqr::proto::pq_ratchet as pb;

// Finding A: PolyEncoder::from_pb accepts per-poly point counts outside {0,1,3,5,30,34,36}.
#[test]
fn encoder_from_pb_inadmissible_len() {
    // 2 points per poly (4 bytes) -- not an admissible V1 length.
    let p = pb::PolynomialEncoder { idx: 0, pts: vec![vec![0u8; 4]; 16], polys: vec![] };
    let mut e = PolyEncoder::from_pb_test(p).expect("from_pb accepted it");
    let _c: Chunk = e.next_chunk(); // idx 0 cache hit
    let _c: Chunk = e.next_chunk(); // idx 1 cache hit
    let c: Chunk = e.next_chunk(); // idx 2 -> cache miss -> from_complete_points(len 2)
    println!("chunk 2 = {:?}", c.data);
}

// Finding B: PolyDecoder::from_pb accepts unbounded point sets per poly.
#[test]
fn decoder_from_pb_too_many_points() {
    let mut slot = Vec::new();
    for x in 0u16..200 {
        slot.extend_from_slice(&x.to_be_bytes());
        slot.extend_from_slice(&[1, 2]);
    }
    let p = pb::PolynomialDecoder { pts_needed: 16 * 100, polys: 16, is_complete: false, pts: vec![slot; 16] };
    let d = PolyDecoder::from_pb_test(p).expect("from_pb accepted it");
    let m = d.decoded_message();
    println!("decoded len = {:?}", m.map(|v| v.len()));
}

// Finding C: pts_needed vs stored pts mismatch -> decoded_message interpolation path with
// more points than MAX_INTERMEDIATE_POLYNOMIAL_DEGREE_V1.
#[test]
fn decoder_from_pb_interpolate_oversized() {
    let mut slot = Vec::new();
    // points at x = 1..=100 (x=0 missing so binary_search misses -> lagrange_interpolate on 100 pts)
    for x in 1u16..=100 {
        slot.extend_from_slice(&x.to_be_bytes());
        slot.extend_from_slice(&[1, 2]);
    }
    let p = pb::PolynomialDecoder { pts_needed: 16 * 100, polys: 16, is_complete: false, pts: vec![slot; 16] };
    let d = PolyDecoder::from_pb_test(p).expect("from_pb accepted it");
    let m = d.decoded_message();
    println!("decoded len = {:?}", m.map(|v| v.len()));
}
