    #[test]
    fn q3_probe_top_level_recv_reaches_panic() {
        // Full production path: spqr::recv(state_blob, msg) with a corrupted chain blob.
        use crate::{Params, Version};
        use prost::Message as _;
        let mk = |direction| {
            crate::initial_state(Params {
                version: Version::V1,
                min_version: Version::V1,
                direction,
                auth_key: &[41u8; 32],
                chain_params: crate::ChainParams::default(),
            })
            .unwrap()
        };
        let a = mk(Direction::A2B);
        let b = mk(Direction::B2A);
        let mut rng = rand::rngs::OsRng.unwrap_err();
        // Exchange enough messages to get a Chain into both states.
        let s1 = crate::send(&a, &mut rng).unwrap();
        let r1 = crate::recv(&b, &s1.msg).unwrap();
        let s2 = crate::send(&r1.state, &mut rng).unwrap();
        let r2 = crate::recv(&s1.state, &s2.msg).unwrap();
        let a = r2.state;
        // Corrupt A's persisted chain blob: every recv.next -> 1 byte.
        let mut st = crate::proto::pq_ratchet::PqRatchetState::decode(a.as_slice()).unwrap();
        let chain = st.chain.as_mut().expect("chain present after exchange");
        for l in chain.links.iter_mut() {
            l.recv.as_mut().unwrap().next = vec![0u8];
        }
        let corrupted = st.encode_to_vec();
        // B sends the next message; A receives with the corrupted blob.
        let s3 = crate::send(&s2.state, &mut rng).unwrap();
        let r = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| crate::recv(&corrupted, &s3.msg)));
        assert!(r.is_err(), "expected panic through spqr::recv");
    }

    #[test]
    fn q4_probe_misaligned_large_prev_forward_path_hits_gc_slice_index() {
        // gc has no alignment assert; misaligned data large enough to trigger GC
        // reaches self.data[i..i+4] with i+4 > len.
        let mut pb = q4_chain().into_pb();
        let d = pb.links[0].recv.as_mut().unwrap();
        d.ctr = 5000;

        assert!(r.is_err());
    }
